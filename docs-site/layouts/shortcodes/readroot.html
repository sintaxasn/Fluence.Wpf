{{- /* readroot - inline a markdown file from the repository root. Parameters file Repository-relative path of the file to include (e.g. "CHANGELOG.md" or "KNOWN_ISSUES.md"). The file must be exposed through the assets/_root mount declared in hugo.toml. skipFirstHeading Optional. When "true", the leading H1 heading is stripped before rendering so the page's own front-matter title is the only document title. Usage: {{< readroot file="CHANGELOG.md" skipFirstHeading="true" >}} */ -}} {{- $file := .Get "file" -}} {{- $skip := eq (.Get "skipFirstHeading") "true" -}} {{- $resource := resources.Get (printf "_root/%s" $file) -}} {{- if $resource -}} {{- $content := $resource.Content -}} {{- if $skip -}} {{- $content = replaceRE `^\s*#\s+[^\n]+\n+` "" $content -}} {{- end -}} {{- $.Page.RenderString $content -}} {{- else -}}
Missing source: {{ $file }} could not be loaded. Confirm the file is mounted under assets/_root in hugo.toml.
{{- end -}}
